<body>
  <div id="cesiumContainer"></div>

  <script>
    const viewer = new Cesium.Viewer("cesiumContainer", {
      terrain: Cesium.Terrain.fromWorldTerrain()
    });

    viewer.camera.flyTo({
      destination: Cesium.Cartesian3.fromDegrees(9.105, 48.745, 1500),
      orientation: {
        heading: Cesium.Math.toRadians(0),
        pitch: Cesium.Math.toRadians(-45),
        roll: 0
      }
    });

    Cesium.GeoJsonDataSource.load("data/observed_buildings_merged.geojson", {
      clampToGround: false
    }).then(function(dataSource) {
      viewer.dataSources.add(dataSource);

      const entities = dataSource.entities.values;
      for (let i = 0; i < entities.length; i++) {
        const e = entities[i];

        e.polygon.material = Cesium.Color.RED.withAlpha(0.6);
        e.polygon.outline = true;
        e.polygon.outlineColor = Cesium.Color.BLACK;
        e.polygon.extrudedHeight = 20;

        e.description =
          "<h3>Observed Building</h3>" +
          "<p><b>Condition:</b> " + (e.properties.condition || "N/A") + "</p>" +
          "<p><b>Notes:</b> " + (e.properties.notes || "N/A") + "</p>" +
          "<p><b>CityObject ID:</b> " + (e.properties.cityobject_id || "N/A") + "</p>";
      }
    });
  </script>
